 73    def _build_nearest_neighbors(self) -> None:
 74        """Populate ``_nearest_neighbors`` and ``_nn_bond_vectors_cache`` together."""
 75        Lx, Ly = self.Lx, self.Ly
 76        neighbors: list[list[int]] = []
 77        bond_vecs: list[NDArray[np.floating]] = []
 78
 79        for index in range(Lx * Ly):
 80            row = index // Lx
 81            col = index % Lx
 82
 83            if (row + col) % 2 == 0:
 84                continue  # emit each (A, B) pair once
 85
 86            for dr, dc in [(-1, 0), (0, -1), (0, 1)]:
 87                nrow, ncol = row + dr, col + dc
 88
 89                # periodic wrap flags (±1 if we crossed a boundary, else 0)
 90                wrap_r = wrap_c = 0
 91                if not (0 <= nrow < Ly):
 92                    if not self.pbc_y:
 93                        continue
 94                    wrap_r = -1 if nrow < 0 else 1
 95                    nrow %= Ly
 96                if not (0 <= ncol < Lx):
 97                    if not self.pbc_x:
 98                        continue
 99                    wrap_c = -1 if ncol < 0 else 1
100                    ncol %= Lx
101
102                neighbor_index = nrow * Lx + ncol
103                neighbors.append([index, neighbor_index])
104
105                # Short bond vector: neighbour position in the periodic image
106                r_i = self.index_to_position(index)
107                r_j = self.index_to_position(neighbor_index)
108                r_j_image = r_j + np.array(
109                    [
110                        wrap_c * Lx * self._col_width,
111                        wrap_r * Ly * self._row_height,
112                    ],
113                    dtype=B.FCPUDTYPE,
114                )
115                bond_vecs.append(r_j_image - r_i)
116
117        self._nearest_neighbors = np.array(neighbors, dtype=int)
118        self._nn_bond_vectors_cache = np.array(bond_vecs, dtype=B.FCPUDTYPE)

168    def _build_nearest_plaquette_neighbors(self) -> None:
169        """Populate ``_nearest_plaquette_neighbors`` and ``_nn_plaquette_vectors_cache`` together.
170        Return bonds corresponding to vortex current between two neighboring plaquettes
171        """
172        neighbors: list[list[int]] = []
173        plaq_boundaries: list[tuple] = []
174        bond_vecs: list[NDArray[np.floating]] = []

176            self._list_plaquettes()
177
178        list_of_plaquettes = self._plaquette_indices[0]
179        # Extract the length of the plaquette-system along the x-direction
180        plaq_Lx = (
181            B.xp().max(self._plaquette_positions[:, 0])
182            - B.xp().min(self._plaquette_positions[:, 0])
183            + self._plaq_xstep
184        )
185        # Extract the length of the plaquette-system along the y-direction
186        plaq_Ly = (
187            B.xp().max(self._plaquette_positions[:, 1])
188            - B.xp().min(self._plaquette_positions[:, 1])
189            + self._plaq_ystep
190        )
191        # Construct x/y-distance and y-distance matrix
192        x_coords = self._plaquette_positions[:, 0].reshape(-1, 1)
193        y_coords = self._plaquette_positions[:, 1].reshape(-1, 1)
194        # Subtract the x/y-coordinates as a (1, n_sites) row vector
195        # array[i, j] = x/y_coords_row[j] - x/y_coords_col[i]
196        x_dist_matrix = x_coords.T - x_coords
197        y_dist_matrix = y_coords.T - y_coords
198        if self.pbc_x:
199            x_dist_matrix -= plaq_Lx * np.round(x_dist_matrix / plaq_Lx)
200        if self.pbc_y:
201            y_dist_matrix -= plaq_Ly * np.round(y_dist_matrix / plaq_Ly)
202        for idx, plaq in enumerate(list_of_plaquettes[:-1], start=0):
203            fwd_adj_plaq_pairs, back_adj_plaq_pairs = get_valid_adjacent_pairs(plaq)
204            for idx2, plaq2 in enumerate(list_of_plaquettes[idx + 1 :], start=idx + 1):
205                fwd_adj_plaq2_pairs, _ = get_valid_adjacent_pairs(plaq2)
206                has_common = not set(back_adj_plaq_pairs).isdisjoint(
207                    fwd_adj_plaq2_pairs
208                )
209                if has_common:
210                    neighbors.append([idx, idx2])
211                    intersecting_boundary = list(
212                        set(back_adj_plaq_pairs).intersection(fwd_adj_plaq2_pairs)
213                    )[0]
214                    idx_intersection_boundary = back_adj_plaq_pairs.index(
215                        intersecting_boundary
216                    )
217                    plaq_boundaries.append(
218                        fwd_adj_plaq_pairs[idx_intersection_boundary]
219                    )
220                    bond_vecs.append(
221                        [x_dist_matrix[idx, idx2], y_dist_matrix[idx, idx2]]
222                    )
223
224        self._nearest_plaquette_neighbors = np.array(neighbors, dtype=int)
225        self._boundary_nearest_plaquette_neighbors = np.array(
226            plaq_boundaries, dtype=int
227        )

230    def _list_plaquettes(self) -> None:
231        """List all plaquettes as lists of site indices in CCW order.
232        Includes boundary pseudo-plaquettes padded with -1 in their strict geometric slots
233        to rigorously preserve physical edge orientations.
234        """
235        Lx, Ly = self.Lx, self.Ly
236        pbc_x, pbc_y = self.pbc_x, self.pbc_y
237
238        # Expand anchor boundaries slightly to catch partial faces on open edges
239        ax_min = 0 if pbc_x else -2
240        ax_max = Lx - 1
241        ay_min = 0 if pbc_y else -1
242        ay_max = Ly - 1
243
244        ax_grid, ay_grid = np.meshgrid(
245            np.arange(ax_min, ax_max + 1), np.arange(ay_min, ay_max + 1)
246        )
247        mask = (ax_grid + ay_grid) % 2 == 0
248        ax = ax_grid[mask]
249        ay = ay_grid[mask]
250
251        path_offsets = self.plaquette_path_offsets_ccw.astype(np.int64)
252        i_offsets = path_offsets[:, 0]
253        i_dr = i_offsets // Lx
254        i_dc = i_offsets % Lx
255
256        rows_cpu_list = []
257        cols_cpu_list = []
258        # anchors_list = []
259        positions_list = []
260
261        for x, y in zip(ax, ay):
262            nodes = []
263            for n_idx in range(6):
264                r = y + i_dr[n_idx]
265                c = x + i_dc[n_idx]
266
267                # Check bounds and apply PBC wrapping
268                if pbc_x:
269                    c = c % Lx
270                elif not (0 <= c < Lx):
271                    nodes.append(-1)
272                    continue
273
274                if pbc_y:
275                    r = r % Ly
276                elif not (0 <= r < Ly):
277                    nodes.append(-1)
278                    continue
279
280                nodes.append(r * Lx + c)
281
282            # A plaquette requires at least 2 valid boundary sites
283            valid_count = sum(1 for n in nodes if n != -1)
284            if valid_count >= 2:
285                # Assign u and v by strictly retaining topological slots.
286                # This guarantees edges automatically inherit the correct CCW sequence.
287                edges_u = nodes
288                edges_v = [nodes[(i + 1) % 6] for i in range(6)]
289
290                rows_cpu_list.append(edges_u)
291                cols_cpu_list.append(edges_v)
292                # anchors_list.append(y * Lx + x)
293
294                # Compute the ideal geometric center directly from the anchor coordinate
295                center_x = float((x + 1) * self._col_width)
296                center_y = float((y + 0.5) * self._row_height)
297                positions_list.append([center_x, center_y])
298
299        rows_cpu = np.array(rows_cpu_list, dtype=np.int64)
300        cols_cpu = np.array(cols_cpu_list, dtype=np.int64)
301
302        self._plaquette_indices = B.xp().array([rows_cpu, cols_cpu], dtype=B.xp().int64)
303        self._plaquette_positions = B.xp().array(positions_list, dtype=B.FCPUDTYPE)
304
305        unique_x = B.xp().unique(self._plaquette_positions[:, 0])
306        self._plaq_xstep = float(B.xp().diff(unique_x)[0]) if len(unique_x) > 1 else 0.0
307
308        unique_y = B.xp().unique(self._plaquette_positions[:, 1])
309        self._plaq_ystep = float(B.xp().diff(unique_y)[0]) if len(unique_y) > 1 else 0.0

421    def index_to_position(self, index: int) -> B.FCPUArray:
422        row = index // self.Lx
423        col = index % self.Lx
424
425        y_offset = 0.25 * (-1) ** ((col + row) % 2)
426
427        x = self._col_width * (index % self.Lx)

 70    """Compute the magnetic field for many observation points at once.
 71
 72    Args:
 73        r_obs_array: array-like of shape (N_obs, 3), points in units of a_NN.
 74        currents: array-like of shape (N_bonds,) in units of one bond-current quantum.
 75        geometry: Lattice2DGeometry instance defining the lattice structure.
 76        n_images: image range for periodic replication. Uses shifts from
 77            -n_images to +n_images along periodic axes.
 78        block_size: number of observation points processed per chunk.
 79        a_nn_m: nearest-neighbor distance in meters. Used to convert geometry units.
 80    Returns:
 81        np.ndarray of shape (N_obs, 3) in Tesla.
 82
 83    Unit trace:
 84        ``r = a_NN * r_hat`` and ``J = I_0 * J_hat`` with
 85        ``I_au = e * E_h / hbar``.
 86        The Biot-Savart kernel is evaluated on the dimensionless positions
 87        ``r_hat`` and then rescaled by ``mu_0 / (4 pi) * I_au / a_NN``.
 88        The final field is therefore returned in Tesla.
 89    """
 90    r_obs_array = np.asarray(r_obs_array, dtype=float)
 91    if r_obs_array.ndim == 1:
 92        r_obs_array = r_obs_array[np.newaxis, :]
 93    if r_obs_array.shape[1] != 3:
 94        raise ValueError("r_obs_array must have shape (N_obs, 3)")
 95
 96    r_i, r_k, J = geometry.prepare_3d_current_segments(
 97        np.asarray(currents, dtype=float),
 98        n_images=n_images,
 99        current_threshold=1e-10,
100    )
101
102    n_obs = r_obs_array.shape[0]
103    if J.size == 0:
104        return np.zeros((n_obs, 3), dtype=float)
105
106    mu0_4pi = 1e-7
107    I_au = current_unit_amperes()
108
109    if a_nn_m is None:
110        a_nn_m = DEFAULT_A_NN_M
111
112    # Dimensionless spatial components require 1 / a_nn correction
113    prefactor = mu0_4pi * I_au / a_nn_m
114
115    B_total = np.zeros((n_obs, 3), dtype=float)
116
117    for start in range(0, n_obs, block_size):
118        stop = min(start + block_size, n_obs)
119        r = r_obs_array[start:stop, np.newaxis, :]
120
121        u = r - r_i[np.newaxis, :, :]
122        v = r - r_k[np.newaxis, :, :]
123
124        len_u = np.linalg.norm(u, axis=2)
125        len_v = np.linalg.norm(v, axis=2)
126        cross_uv = np.cross(u, v, axis=2)
127        norm_cross_sq = np.sum(cross_uv**2, axis=2)
128        uv = np.sum(u * v, axis=2)
129
130        denom = len_u * len_v
131        with np.errstate(divide="ignore", invalid="ignore"):
132            scalar_term = (len_u + len_v) * (1.0 - (uv / denom))
133
134        valid_obs = (norm_cross_sq > 1e-18) & (denom > 1e-18)
135        with np.errstate(divide="ignore", invalid="ignore"):
136            coeff = np.where(
137                valid_obs,
138                prefactor * J[np.newaxis, :] * scalar_term / norm_cross_sq,
139                0.0,
140            )
141
142        B_segments = coeff[:, :, np.newaxis] * cross_uv
143        B_total[start:stop] = np.sum(B_segments, axis=1)
144
145    return B_total

13def fourier_at_omega(signal: B.FCPUArray, dt: float, omega: float) -> np.ndarray:
14    """
15    Calculates the discrete Fourier component at omega for a signal of arbitrary shape.
16    Assumes axis 0 is the time dimension.
17    """
18    arr: np.ndarray = np.asarray(signal)
19    
20    N = len(arr)
21    t = np.arange(N) * dt
22    phase = np.exp(-1j * omega * t)
23    
24    # Contract axis 0 (time) of signal with axis 0 of phase
25    # Equivalent to: sum(signal[t, ...] * phase[t] for t)
26    return np.tensordot(arr, phase, axes=([0], [0])) * dt

331    if not cbar_axes:
332        # First colorbar: align top with main ax, pad right
333        cax_x = min(0.98 - cbar_w, ax_bbox.x1 + 0.02)
334        cax_y = min(0.95 - cbar_h, ax_bbox.y0 + ax_bbox.height - cbar_h)
335        lgd = ax.get_legend()
336        if lgd:
337            # Shift the first colorbar below the legend if it exists
338            lgd_bbox = lgd.get_window_extent().transformed(fig.transFigure.inverted())
339            cax_x = lgd_bbox.x0
340            cax_y = lgd_bbox.y0 - 0.02 - cbar_h
341        cax_y = max(0.05, cax_y)
342    else:
343        if direction == "vertical":
344            # Stack below the lowest colorbar in the grid
345            min_y0 = min(a.get_position().y0 for a in cbar_axes)
346            # Re-use the starting x-coordinate of the first appended colorbar
347            first_x0 = cbar_axes[0].get_position().x0
348            cax_x = first_x0
349            cax_y = max(0.05, min_y0 - cbar_spacing_y - cbar_h)
350        elif direction == "horizontal":
351            # Stack to the right of the *last* appended colorbar
352            last_cax_bbox = cbar_axes[-1].get_position()
353            cax_x = last_cax_bbox.x1 + cbar_spacing_x
354            cax_y = last_cax_bbox.y0
355        else:
356            raise ValueError("direction must be 'vertical' or 'horizontal'")
357
358    cax = fig.add_axes((cax_x, cax_y, cbar_w, cbar_h), label="appended_colorbar")
359    cb = fig.colorbar(mappable, cax=cax, orientation="vertical", **kwargs)
360    if label:
361        cb.set_label(label, size="small")
362    cb.ax.tick_params(labelsize="small")
363    return cb

Append a colorbar to the figure dynamically using absolute positioned axes.

This avoids stealing space from the main axes (which can ruin 'equal' aspect plots). Subsequent calls to this function will stack colorbars in a grid format.

Args: fig: The matplotlib Figure. ax: The main matplotlib Axes. mappable: The ScalarMappable to draw the colorbar for. label: Optional label for the colorbar. direction: 'vertical' (starts a new row below the lowest colorbar) or 'horizontal' (starts a new column to the right of the last appended colorbar). colorbar_width: Optional absolute colorbar width in figure coordinates. If omitted and previous appended colorbars exist, their width is reused. colorbar_height: Optional absolute colorbar height in figure coordinates. If omitted and previous appended colorbars exist, their height is reused. **kwargs: Additional keyword arguments passed to fig.colorbar().

Returns: The created Colorbar instance.

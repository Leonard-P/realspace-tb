 12def compute_A_to_Bz_x(x_values, y_values, bz):
 13    """Compute $$\frac{\int x B_{z}(x,y)dxdy}{\int |B_{z}(x,y)|dxdy}$$
 14    Parameters:
 15        x_values: 1D array of x coordinates corresponding to the bz values.
 16        y_values: 1D array of y coordinates corresponding to the bz values.
 17        bz: 2D array of Bz values on the grid defined by x_values and y_values."""
 18    dx = x_values[1] - x_values[0]
 19    dy = y_values[1] - y_values[0]
 20    dxdy = dx * dy
 21
 22    numerator = np.sum(x_values * bz) * dxdy
 23    denominator = np.sum(np.abs(bz)) * dxdy
 24
 25    return numerator / denominator if denominator != 0 else 0

 28class VorticityObservable(Observable):
 29    r"""Measures the vorticity at each smallest possible plaquette
 30
 31    $$\sum_{(R,R')\in\partial P} \hat{J}_{R\rightarrow R'}$$
 32    """
 33
 34    def __init__(
 35        self,
 36        geometry: HoneycombLatticeGeometry,
 37        window: MeasurementWindow | None = None,
 38        hamiltonian: Hamiltonian | None = None,
 39    ):
 40        super().__init__(window)
 41
 42        if hamiltonian is None:
 43            print(
 44                f"No Hamiltonian passed to {self.__class__}. Assuming Onsite-Potential Hamiltonian with t_hop=1."
 45            )
 46
 47        self._hamiltonian = hamiltonian
 48        # changing the prefactor to calculate vorticity instead
 49        self._c = 1.0  # -electron_mass * geometry.plaquette_area / 3
 50        plaquette_indices, _, _, _ = geometry.plaquettes
 51        self._rows = plaquette_indices[0]
 52        self._cols = plaquette_indices[1]
 53
 54    def _compute_edge_currents(self, rho: B.Array, t: float) -> B.Array:
 55        r"""Compute bond currents along plaquette edges.
 56
 57        Uses the gauge-invariant formula
 58        $I_{i<-j}(t) = 2\,\mathrm{Im}(H_{ij}(t)\,\rho_{ji}(t))$.
 59        When no Hamiltonian is stored (`t_{hop} = -1`), this reduces to
 60        $2\,\mathrm{Im}(\rho_{ij})$.
 61        """
 62        invalid_mask = (self._rows == -1) | (self._cols == -1)
 63        if self._hamiltonian is not None:
 64            H_t = self._hamiltonian.at_time(t)
 65            xp = B.xp()
 66            rows_flat = self._rows.ravel()
 67            cols_flat = self._cols.ravel()
 68            # Order reversed due to sign-change from -1.0 to +1.0 in self._c, which flips the current direction convention and thus the order of indices in the current formula. This is a bit subtle and could be made clearer by defining a helper function for the current that takes care of the index ordering and sign convention.
 69            # h_ij = xp.asarray(H_t[rows_flat, cols_flat]).reshape(self._rows.shape)
 70            h_ij = xp.asarray(H_t[cols_flat, rows_flat]).reshape(self._cols.shape)
 71            rho_values = rho[self._rows, self._cols]
 72            rho_values = B.xp().where(invalid_mask, 0.0, rho_values)
 73            # return 2.0 * xp.imag(h_ij * rho[self._cols, self._rows])
 74            return 2.0 * xp.imag(h_ij * rho_values)
 75        rho_values = rho[self._cols, self._rows]
 76        rho_values = B.xp().where(invalid_mask, 0.0, rho_values)
 77        # return 2.0 * xp.imag(h_ij * rho[self._rows, self._cols])
 78        return 2.0 * B.xp().imag(rho_values)
 79
 80    def _compute_vortices(self, rho: B.Array, t: float) -> B.Array:
 81        I_edges = self._compute_edge_currents(rho, t)
 82        return B.xp().sum(I_edges, axis=1)  # (n_cells,)
 83
 84    def _compute(self, rho: B.Array, t: float) -> B.Array:
 85        return self._c * self._compute_vortices(rho, t)

 88class VortFluxObservable(VorticityObservable):
 89    r"""Measures the unmodified vorticity flux
 90
 91    $$\hat{\Tilde{j}}^{z}_{\omega,P_{i}\rightarrow P_{j}}
 92    =(e_{z}\cross n_{P_{i} -> P})\cdot\sum_{X}\langle\hat{\Pi}^{(b)}_{X}\rangle(t)$$
 93    This observable is ANTISYMMETRIC, since n_{P_{i} -> P} = -n_{P -> P_{i}}
 94    1. WARNING: Currently hard-coded for 2D lattices, whose plaquettes have all their normal
 95                vectors pointing along one (z-)direction.
 96    """
 97
 98    def __init__(
 99        self,
100        geometry: HoneycombLatticeGeometry,
101        window: MeasurementWindow | None = None,
102        hamiltonian: Hamiltonian | None = None,
103    ):
104        super().__init__(geometry, window, hamiltonian)
105        self._nearest_plaquette_neighbors = geometry.nearest_plaquette_neighbors
106        self._boundary_plaquette_vectors = geometry.boundary_nn_plaquette_vectors
107        self._num_plaquettes, _ = self._rows.shape
108
109    def _compute_div_edge_fluxes_per_bond(self, rho: B.Array, t: float) -> B.Array:
110        r"""Measures divergent of the current flux tensor for each bond at
111        the intersection of two neighboring plaquettes $$\sum_{X}\langle\hat{\Pi}^{R->R'}_{X}\rangle(t)$$
112        """
113        r_arr = B.xp().array(self._boundary_plaquette_vectors[:, 0], dtype=B.xp().int64)
114        rp_arr = B.xp().array(
115            self._boundary_plaquette_vectors[:, 1], dtype=B.xp().int64
116        )
117        assert len(r_arr) == len(
118            rp_arr
119        ), "Nearest neighbor list should have shape (n_edges, 2)."
120        if self._hamiltonian is not None:
121            xp = B.xp()
122            H_t = self._hamiltonian.at_time(t)
123            h_scalars = xp.asarray(H_t[rp_arr, r_arr]).ravel()
124            H_rho = H_t.dot(rho)
125            rho_H = rho @ H_t
126            H_rho_vals = H_rho[r_arr, rp_arr]
127            rho_H_vals = rho_H[r_arr, rp_arr]
128            return 2.0 * xp.real(h_scalars * (H_rho_vals - rho_H_vals))
129        return 2.0 * B.xp().real(rho_H_vals - H_rho_vals)
130
131    def _compute_tilde_j_omega(self, rho: B.Array, t: float) -> B.Array:
132        div_edge_fluxes_per_bond = self._compute_div_edge_fluxes_per_bond(rho, t)
133        tilde_j_omega = B.xp().zeros(
134            (self._num_plaquettes, self._num_plaquettes),
135            dtype=div_edge_fluxes_per_bond.dtype,
136        )
137        i_indices = B.xp().array(
138            self._nearest_plaquette_neighbors[:, 0], dtype=B.xp().int64
139        )
140        j_indices = B.xp().array(
141            self._nearest_plaquette_neighbors[:, 1], dtype=B.xp().int64
142        )
143
144        tilde_j_omega[i_indices, j_indices] = div_edge_fluxes_per_bond
145        tilde_j_omega[j_indices, i_indices] = -div_edge_fluxes_per_bond
146        return tilde_j_omega
147
148    def _compute(self, rho: B.Array, t: float) -> B.Array:
149        return self._compute_tilde_j_omega(rho, t)

152class VortFluxModObservable(VortFluxObservable):
153    r"""Measures the modified vorticity flux
154
155    $$\hat{j}^{z}_{\omega,P_{i}\rightarrow P_{j}}
156    =(e_{z}\cross n_{P_{i} -> P})\cdot\biggl(\langle\sum_{X}\hat{\Pi}^{(b)}_{X}\rangle(t)
157    -\langle\hat{f}^{(b)}\rangle(t)\biggr)$$
158    This observable is ANTISYMMETRIC, since n_{P_{i} -> P} = -n_{P -> P_{i}}
159    1. WARNING: Currently hard-coded for 2D lattices, whose plaquettes have all their normal
160                vectors pointing along one (z-)direction.
161    """
162
163    def __init__(
164        self,
165        geometry: HoneycombLatticeGeometry,
166        window: MeasurementWindow | None = None,
167        hamiltonian: Hamiltonian | None = None,
168    ):
169        super().__init__(geometry, window, hamiltonian)
170        self._nearest_plaquette_neighbors = geometry.nearest_plaquette_neighbors
171        self._boundary_plaquette_vectors = geometry.boundary_nn_plaquette_vectors
172        self._num_plaquettes, _ = self._rows.shape
173
174    def _compute_edge_forces_per_bond(self, rho: B.Array, t: float) -> B.Array:
175        r"""Measures forces for each bond at the intersection of two neighboring plaquettes
176        $$\sum_{X}\langle\hat{\Pi}^{R->R'}_{X}\rangle(t)$$"""
177        r_arr = B.xp().array(self._boundary_plaquette_vectors[:, 0], dtype=B.xp().int64)
178        rp_arr = B.xp().array(
179            self._boundary_plaquette_vectors[:, 1], dtype=B.xp().int64
180        )
181        assert len(r_arr) == len(
182            rp_arr
183        ), "Nearest neighbor list should have shape (n_edges, 2)."
184        if self._hamiltonian is not None:
185            xp = B.xp()
186            dH_dt = self._hamiltonian.derivative_at_time(t)
187            h_scalars = xp.asarray(dH_dt[rp_arr, r_arr]).ravel()
188            # return 2.0 * xp.real(h_scalars * (H_rho_vals - rho_H_vals))
189            return 2.0 * xp.imag(h_scalars * rho[r_arr, rp_arr])
190        return 2.0 * B.xp().imag(rho[rp_arr, r_arr])
191
192    def _compute_f_j_omega(self, rho: B.Array, t: float) -> B.Array:
193        edge_forces_per_bond = self._compute_edge_forces_per_bond(rho, t)
194        f_j_omega = B.xp().zeros(
195            (self._num_plaquettes, self._num_plaquettes),
196            dtype=edge_forces_per_bond.dtype,
197        )
198        i_indices = B.xp().array(
199            self._nearest_plaquette_neighbors[:, 0], dtype=B.xp().int64
200        )
201        j_indices = B.xp().array(
202            self._nearest_plaquette_neighbors[:, 1], dtype=B.xp().int64
203        )
204
205        f_j_omega[i_indices, j_indices] = edge_forces_per_bond
206        f_j_omega[j_indices, i_indices] = -edge_forces_per_bond
207        return -f_j_omega
208
209    def _compute(self, rho: B.Array, t: float) -> B.Array:
210        return self._compute_tilde_j_omega(rho, t) + self._compute_f_j_omega(rho, t)

270class VortPolObservable(VorticityObservable):
271    r"""Measures the vortex polarization
272    Boundary pseudo-plaquettes padded with -1 in their strict geometric slots are included
273
274    $$\mu^{i}_{\omega}=\sum_{k}\frac{\Omega_{k}}{A_{k}}\sum_{(R,R')\in\partial P_{k}}J_{R\rightarrow R'}(t)
275    |R'-R|\hat{n}^{k}_{i}(R^{k}+\alpha^{k}-R^{ref})$$
276    \alpha^{k} is chosen such that R^{k}+\alpha^{k} points to the center of the plaquette k.
277    1. WARNING: Currently hard-coded for 2D lattices, whose plaquettes have all their normal
278                vectors pointing along one (z-)direction and all equal surface area.
279    2. WARNING: Observable is well-defined only along the direction of the OBC.!
280    """
281
282    def __init__(
283        self,
284        geometry: HoneycombLatticeGeometry,
285        window: MeasurementWindow | None = None,
286        hamiltonian: Hamiltonian | None = None,
287    ):
288
289        super().__init__(geometry, window, hamiltonian)
290        if not isinstance(geometry, Lattice2DGeometry):
291            raise TypeError(
292                "Currently supports only Lattice2DGeometry with all plaquettespointing in the same direction."
293            )
294        if geometry.pbc_x and geometry.pbc_y:
295            raise ValueError(
296                "Observable is well-defined only along the direction of the OBC. Please ensure that at least one of the directions has OBC."
297            )
298        _, plaquette_positions, _, _ = geometry.plaquettes
299        origin = B.xp().mean(plaquette_positions, axis=0)
300        # Put the origin of CF in the centre of the current vortices
301        self._plaq_pos = plaquette_positions - origin
302        # Total area of plaquettes of the system
303        self._A = plaquette_positions.shape[0] * geometry.plaquette_area
304
305    def _compute(self, rho: B.Array, t: float) -> B.Array:
306        # Calculate vorticity in each plaquette
307        vortices = self._c * self._compute_vortices(rho, t)
308        # Multiply vortices with plaquette positions
309        vort_times_r = self._plaq_pos * vortices[:, np.newaxis]
310
311        return B.xp().sum(vort_times_r, axis=0) / self._A

362class VectorVortFluxObservable(VortFluxObservable):
363    r"""Measures the vector-valued unmodified vortex flow
364
365    $$\Tilde{\mathbf{j}}^{z}_{\omega,P_{i}\rightarrow P_{j}}
366    =1/2\sum_{i,j}\Bigl(\hat{\Tilde{j}}^{z}_{\omega,P_{i}\rightarrow P_{j}}(R_{j}-R_{i})\Bigr)$$
367    WARNING: Currently hard-coded for 2D lattices, whose plaquettes have all their normal
368             vectors pointing along one (z-)direction and all equal surface area.
369    """
370
371    def __init__(
372        self,
373        geometry: HoneycombLatticeGeometry,
374        window: MeasurementWindow | None = None,
375        hamiltonian: Hamiltonian | None = None,
376    ):
377
378        super().__init__(geometry, window, hamiltonian)
379        assert isinstance(
380            geometry, Lattice2DGeometry
381        ), "Currently supports only Lattice2DGeometry with all plaquettes pointing in the same direction."
382        # Take the position of each plaquette, difference in x & y between two nearest neighboring plaquettes
383        _, plaquette_positions, plaq_xstep, plaq_ystep = geometry.plaquettes
384        # Extract the length of the plaquette-system along the x-direction
385        self._plaq_Lx = (
386            B.xp().max(plaquette_positions[:, 0])
387            - B.xp().min(plaquette_positions[:, 0])
388            + plaq_xstep
389        )
390        # Extract the length of the plaquette-system along the y-direction
391        self._plaq_Ly = (
392            B.xp().max(plaquette_positions[:, 1])
393            - B.xp().min(plaquette_positions[:, 1])
394            + plaq_ystep
395        )
396        # Total area of plaquettes of the system
397        self._A = plaquette_positions.shape[0] * geometry.plaquette_area
398        # Construct x/y-distance and y-distance matrix
399        x_coords = plaquette_positions[:, 0].reshape(-1, 1)
400        y_coords = plaquette_positions[:, 1].reshape(-1, 1)
401        # Subtract the x/y-coordinates as a (1, n_sites) row vector
402        # array[i, j] = x/y_coords_row[j] - x/y_coords_col[i]
403        self._x_dist_matrix = x_coords.T - x_coords
404        self._y_dist_matrix = y_coords.T - y_coords
405        if geometry.pbc_x:
406            self._x_dist_matrix -= self._plaq_Lx * np.round(
407                self._x_dist_matrix / self._plaq_Lx
408            )
409        if geometry.pbc_y:
410            self._y_dist_matrix -= self._plaq_Ly * np.round(
411                self._y_dist_matrix / self._plaq_Ly
412            )
413
414    def _compute_vector_vort_flux_tensor(
415        self, rho: B.Array, t: float
416    ) -> Tuple[B.Array, B.Array]:
417        tilde_j_omega = self._compute_tilde_j_omega(rho, t)
418        # Omega = 0.5 * (Lambda - Lambda.T)
419        tilde_j_omega_x = 0.5 * tilde_j_omega * self._x_dist_matrix
420        tilde_j_omega_y = 0.5 * tilde_j_omega * self._y_dist_matrix
421        return tilde_j_omega_x, tilde_j_omega_y
422
423    def _compute(self, rho: B.Array, t: float) -> B.Array:
424        tilde_j_omega_x, tilde_j_omega_y = self._compute_vector_vort_flux_tensor(rho, t)
425        tilde_j_omega_x, tilde_j_omega_y = (
426            tilde_j_omega_x.ravel(),
427            tilde_j_omega_y.ravel(),
428        )
429        vec_tilde_j_omega = B.xp().column_stack((tilde_j_omega_x, tilde_j_omega_y))
430        return B.xp().sum(vec_tilde_j_omega, axis=0) / self._A
